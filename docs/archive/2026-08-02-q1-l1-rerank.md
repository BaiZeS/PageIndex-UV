**Goal:** 固化可复现的评测基准，并把 L1 文档选择从"裸 LLM 列表"升级为"粗选 + 显式相关性打分 + 取 top-k"，直接收窄与 Self-RAG 在 Tech/Finance 上的精确率差距，并抑制 run-to-run 抖动。

**Architecture:** 改动集中在 `SuperTreeIndex.select_documents`（`pageindex_mutil/super_tree.py`）。新增 `_rank_k` 配置（默认 12）与一个打分方法：LLM 对粗选候选逐篇输出"相关性分数 + 理由"，按分数排序取前 3-5 篇返回同名 `list[str]`，接口对调用方（`router._search_super_tree`）不变。随后在 `pageindex-paper` 侧固话一个可复现的长档 F1 基线脚本。

**Tech Stack:** Python 3.12 / pytest 8+ / pytest-asyncio（asyncio_mode=auto）/ 既有 `patch.object(llm_acompletion)` 测试模式 / temp SQLite（`PageIndexDB`）/ jieba。

在 `super_tree.py` 中 `select_documents` 之前新增打分方法：

```python
    async def _score_candidates(self, query: str, candidate_db_ids: Dict[int, float]) -> list[str]:
        """Q1：对候选文档做显式相关性打分，返回降序列出的 top-k uuid。

        相较于旧逻辑让 LLM 直接吐 doc_id，这里让 LLM 对每篇给出
        相关性分数与理由，按分数取前 _SELECT_TOP_K，提升难负样本下的精确率。
        """
        if not candidate_db_ids:
            return []

        truncated = self._truncate_candidates(candidate_db_ids)
        if len(truncated) <= 1:
            # 仅一个候选时无需打分，直接返回
            uuid = self._get_db_to_uuid().get(truncated[0], str(truncated[0]))
            return [uuid]

        super_tree = self._build_super_tree(truncated)
        kb_identity = self.kb_identity.get_identity()

        prompt = f"""你是一个文档检索相关性评分专家。给定用户问题、知识库概览和候选文档结构，请为每个候选文档给出 0.0-1.0 的相关性分数。

[知识库概览]
{kb_identity}

[用户问题]
{query}

[候选文档结构]
{json.dumps(super_tree, ensure_ascii=False)}

要求：
1. 分数越高代表文档越可能包含答案；只给 >=0.5 的候选保留，其余可在 ranked 中省略。
2. 用一句话说明理由。

返回JSON格式：
{{
  "ranked": [
    {{"doc_id": "uuid-1", "score": 0.9, "reason": "直接相关"}},
    ...
  ],
  "top_k": {self._SELECT_TOP_K}
}}
直接返回最终JSON结构，不要输出其他内容。"""

        response = await llm_acompletion(self.retrieve_model or self.model, prompt)
        if not response:
            return []

        data = extract_json(response)
        if not isinstance(data, dict):
            return []

        ranked = data.get("ranked", [])
        if not isinstance(ranked, list) or not ranked:
            return []

        scored = []
        db_to_uuid = self._get_db_to_uuid()
        for item in ranked:
            if not isinstance(item, dict):
                continue
            doc_id = item.get("doc_id")
            if not doc_id:
                continue
            try:
                score = float(item.get("score", 0.0))
            except (TypeError, ValueError):
                continue
            scored.append((doc_id, score))

        scored.sort(key=lambda x: x[1], reverse=True)
        try:
            top_k = max(1, min(int(data.get("top_k", self._SELECT_TOP_K)), len(scored)))
        except (TypeError, ValueError):
            top_k = self._SELECT_TOP_K

        return [doc_id for doc_id, _s in scored[:top_k]]
```

新建 `pageindex-paper/scripts/benchmark_piuv.py`，复用现有 `analyze_results.py` 的解析逻辑，固定 seed/model，输出可对比表格。核心骨架：

```python
#!/usr/bin/env python3
"""阶段0基线：跑长档 benchmark 并输出可复现 F1 表。
用法: uv run python3 scripts/benchmark_piuv.py --domain tech --output results/tables
"""
import argparse, json, math, glob, os
from pathlib import Path

def f1_per_query(gt: list, pred: list) -> float:
    gt_s, pred_s = set(gt), set(pred)
    inter = len(gt_s & pred_s)
    if inter == 0:
        return 0.0
    p = inter / len(pred_s) if pred_s else 0.0
    r = inter / len(gt_s) if gt_s else 0.0
    return 2 * p * r / (p + r)

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--domain", choices=["tech", "medical", "legal", "finance"])
    ap.add_argument("--output", default="results/tables")
    args = ap.parse_args()

    test = json.load(open(f"data/{args.domain}_longdoc/test.json", encoding="utf-8"))
    latest = sorted(glob.glob(f"results/pageindex_uv_{args.domain}_longdoc_*.json"))[-1]
    results = json.load(open(latest, encoding="utf-8"))

    f1s = [f1_per_query(q.get("gold", []), r.get("matched_docs", []))
           for q, r in zip(test, results)]
    avg_f1 = sum(f1s) / len(f1s) if f1s else 0.0
    md = f"| Dataset | Avg F1 | Precision | Recall | File |\n|---|---|---|---|---|\n"
    # 打印行（精确率/召回率聚合逻辑与 analyze_results.py 一致，此处示范平均）
    md += f"| {args.domain} | {avg_f1:.3f} | - | - | `{os.path.basename(latest)}` |\n"
    out = Path(args.output)
    out.mkdir(parents=True, exist_ok=True)
    (out / f"piuv_baseline_{args.domain}.md").write_text(md, encoding="utf-8")
    print(md)

if __name__ == "__main__":
    main()
```
